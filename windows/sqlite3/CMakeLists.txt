cmake_minimum_required(VERSION 3.14)
project(sqlite3_local LANGUAGES C CXX)

# Use local SQLite instead of downloading
set(SQLITE3_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sqlite-amalgamation-3520000")

add_library(sqlite3 SHARED
  "${SQLITE3_SOURCE_DIR}/sqlite3.c"
)

target_include_directories(sqlite3 PUBLIC "${SQLITE3_SOURCE_DIR}")
target_compile_options(sqlite3 PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O2>" "/w")

target_compile_definitions(sqlite3 PRIVATE
  SQLITE_ENABLE_DBSTAT_VTAB
  SQLITE_ENABLE_FTS5
  SQLITE_ENABLE_RTREE
  SQLITE_ENABLE_MATH_FUNCTIONS
  SQLITE_DQS=0
  SQLITE_DEFAULT_MEMSTATUS=0
  SQLITE_TEMP_STORE=2
  SQLITE_MAX_EXPR_DEPTH=0
  SQLITE_STRICT_SUBTYPE=1
  SQLITE_OMIT_AUTHORIZATION
  SQLITE_OMIT_DECLTYPE
  SQLITE_OMIT_DEPRECATED
  SQLITE_OMIT_PROGRESS_CALLBACK
  SQLITE_OMIT_SHARED_CACHE
  SQLITE_USE_ALLOCA
  SQLITE_THREADSAFE=1
)

# Copy precompiled DLL to output
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/sqlite3.dll"
  DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)
